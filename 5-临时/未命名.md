cat > /usr/local/bin/httpsplus << 'EOF'
#!/usr/bin/env bash
# ==========================================
# SSL 证书管理面板 (自定义路径版)
# ==========================================

ACME_BIN="/root/.acme.sh/acme.sh"
DATA_DIR="/etc/httpsplus"
CONFIG_FILE="${DATA_DIR}/config.conf"
NGINX_CONF_DIR="/etc/nginx/conf.d"

GREEN='\033[0;32m'
YELLOW='\033[1;33m'
CYAN='\033[0;36m'
RED='\033[0;31m'
NC='\033[0m'

if [ "$EUID" -ne 0 ]; then
  echo -e "${RED}请使用 root 运行此脚本${NC}"
  exit 1
fi

# 初始化配置
mkdir -p "$DATA_DIR"
if [ ! -f "$CONFIG_FILE" ]; then
    echo "CERT_DIR=\"/etc/nginx/ssl\"" > "$CONFIG_FILE"
fi
source "$CONFIG_FILE"

# 每次启动强制切回 Let's Encrypt
"$ACME_BIN" --set-default-ca --server letsencrypt >/dev/null 2>&1

# 申请证书
issue_cert() {
    clear
    echo -e "${CYAN}--- 申请新证书 (DNS 模式) ---${NC}"
    echo -e "当前证书存放路径: ${GREEN}${CERT_DIR}${NC}"
    read -p "请输入域名 (如 example.com): " domain
    [ -z "$domain" ] && return
    
    echo -e "${YELLOW}正在申请，强制等待 120 秒让 DNS 生效...${NC}"
    "$ACME_BIN" --issue --dns dns_ali -d "$domain" --force --dnssleep 120
    
    if [ $? -ne 0 ]; then
        echo -e "${RED}申请失败，请检查 API 密钥或域名解析。${NC}"
        read -n 1 -s -r -p "按任意键返回..."
        return
    fi
    
    # 安装证书
    local d_cert="${CERT_DIR}/${domain}"
    mkdir -p "$d_cert"
    "$ACME_BIN" --install-cert -d "$domain" --ecc \
        --key-file "${d_cert}/privkey.pem" \
        --fullchain-file "${d_cert}/fullchain.pem" \
        --reloadcmd "systemctl reload nginx || /usr/local/nginx/sbin/nginx -s reload" >/dev/null 2>&1

    # 自动生成 Nginx 443 配置文件
    cat > "${NGINX_CONF_DIR}/https_${domain}.conf" <<NGINXEOF
server {
    listen 443 ssl http2;
    server_name ${domain};
    ssl_certificate ${d_cert}/fullchain.pem;
    ssl_certificate_key ${d_cert}/privkey.pem;
    location / {
        root /usr/share/nginx/html;
        index index.html;
    }
}
NGINXEOF
    # 兼容 systemctl 和原生重载
    systemctl reload nginx 2>/dev/null || /usr/local/nginx/sbin/nginx -s reload 2>/dev/null
    echo -e "${GREEN}✅ 证书申请成功并已保存至 ${d_cert}！${NC}"
    read -n 1 -s -r -p "按任意键返回..."
}

# 修改证书路径
change_cert_dir() {
    clear
    echo -e "${CYAN}--- 修改证书存放路径 ---${NC}"
    echo -e "当前路径: ${GREEN}${CERT_DIR}${NC}"
    read -p "请输入新的绝对路径 (如 /usr/local/nginx/conf/ssl): " new_dir
    if [ -n "$new_dir" ]; then
        echo "CERT_DIR=\"${new_dir}\"" > "$CONFIG_FILE"
        export CERT_DIR="$new_dir"
        mkdir -p "$CERT_DIR"
        echo -e "${GREEN}路径已更新为: ${CERT_DIR}${NC}"
    else
        echo -e "${RED}输入不能为空！${NC}"
    fi
    sleep 2
}

# 配置密钥
config_api() {
    clear
    echo -e "${CYAN}--- 配置阿里云 API ---${NC}"
    read -p "请输入 AccessKey ID: " ali_key
    read -p "请输入 AccessKey Secret: " ali_secret
    if [ -n "$ali_key" ] && [ -n "$ali_secret" ]; then
        export Ali_Key="$ali_key"
        export Ali_Secret="$ali_secret"
        echo -e "${GREEN}密钥已在当前会话导入。${NC}"
    else
        echo -e "${RED}输入不能为空！${NC}"
    fi
    sleep 2
}

# 查看证书
check_status() {
    clear
    echo -e "${CYAN}--- 证书列表 (路径: ${CERT_DIR}) ---${NC}"
    printf "%-30s %-15s\n" "域名" "剩余天数"
    echo "----------------------------------------------"
    for d_cert in ${CERT_DIR}/*; do
        if [ -d "$d_cert" ] && [ -f "${d_cert}/fullchain.pem" ]; then
            domain=$(basename "$d_cert")
            end_date=$(openssl x509 -enddate -noout -in "${d_cert}/fullchain.pem" | cut -d= -f2)
            days_left=$(( ($(date -d "$end_date" +%s) - $(date +%s)) / 86400 ))
            printf "%-30s %-15s\n" "$domain" "$days_left 天"
        fi
    done
    echo "----------------------------------------------"
    read -n 1 -s -r -p "按任意键返回..."
}

# 开启定时任务
install_cron() {
    cat > /etc/systemd/system/httpsplus-renew.service <<SERVICEEOF
[Unit]
Description=Renew SSL
After=network.target

[Service]
Type=oneshot
ExecStart=/root/.acme.sh/acme.sh --cron --dnssleep 120
SERVICEEOF
    cat > /etc/systemd/system/httpsplus-renew.timer <<TIMEREOF
[Unit]
Description=Renew SSL every day
[Timer]
OnCalendar=daily
RandomizedDelaySec=3600
Persistent=true
[Install]
WantedBy=timers.target
TIMEREOF
    systemctl daemon-reload
    systemctl enable --now httpsplus-renew.timer
    echo -e "${GREEN}后台自动续签已开启。${NC}"
    read -n 1 -s -r -p "按任意键返回..."
}

# 主菜单
while true; do
    clear
    if systemctl is-active --quiet httpsplus-renew.timer; then 
        c_status="${GREEN}运行中${NC}"
    else 
        c_status="${RED}未启用${NC}"
    fi
    
    echo -e "${CYAN}==========================================${NC}"
    echo -e "${CYAN}       SSL 证书管理面板 (灵活路径版)      ${NC}"
    echo -e "${CYAN}==========================================${NC}"
    echo -e "  默认 CA      : ${GREEN}Let's Encrypt${NC}"
    echo -e "  证书存放路径 : ${YELLOW}${CERT_DIR}${NC}"
    echo -e "  自动续签状态 : $c_status"
    echo -e "${CYAN}------------------------------------------${NC}"
    echo -e "  [1] 配置阿里云 AccessKey"
    echo -e "  [2] 申请新证书并部署到 Nginx"
    echo -e "  [3] 修改证书存放路径 (编译安装适用)"
    echo -e "  [4] 查看已安装证书"
    echo -e "  [5] 开启后台自动续签"
    echo -e "  [0] 退出"
    echo -e "${CYAN}------------------------------------------${NC}"
    read -p " 请选择 [0-5]: " choice
    
    case $choice in
        1) config_api ;;
        2) issue_cert ;;
        3) change_cert_dir ;;
        4) check_status ;;
        5) install_cron ;;
        6) exit 0 ;;
        *) sleep 1 ;;
    esac
done
EOF
